Ansible - Create Docker Container for Multiple Applications

- name: Setup Docker Container for Multiple Apps
  hosts: all
  become: yes
  tasks:
    - name: Pull Web App Image
      docker_image:
        name: my-web-app
        source: pull

    - name: Pull Database App Image
      docker_image:
        name: my-database-app
        source: pull
    
    - name: Run Web App Image
      docker_container:
        name: my-web-app
        image: web-app
        state: started
        published_ports:
          - "8080:80"

    - name: Run Database App container
      docker_container:
        name: database-app
        image: my-database-app
        state: started
        published_ports:
          - "3306:3306:

Terraform - VPC with Private & Public subnet
resource "aws_vpc" "main" {
    cidr_block = "10.0.0.0/16"
}

resource "aws_subnet" "public" {
    vpc_id = aws_vpc.main.vpc_id
    cidr_block = "10.0.1.0/24"
    availability_zone = "us-east-1a"
    map_public_ip_on_launch = true
}

resource "aws_subnet" "private" {
    vpc_id = aws_vpc.main.id
    cidr_block = "10.0.2.0/24"
    availability_zone = "us-east-1b"
}

Kubernetes - Deploying StatefulSet:
apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: pathnex-db
spec:
  serviceName: "pathnex-db"
  replicas: 3
  selector:
    mathLabels:
      app: pathnex-db
  template:
    metadata:
      lables:
        app: pathnex-db
    spec:
      containers:
        - name: db
          image: mysql-storage
          volumeMounts:
            - mountPath: /var/lib/mysql
              name: mysql-storage
volumeClaimTemplates:
  - metadata:
      name: mysql-storage
    spec:
      accessModes: ["ReadWriteOnce"]
      resources:
        requests:
          storage: 1Gi

Jenkinsfile - Build & Deploy to K8s:
pipeline {
    agent any
    environment {
        IMAGE_NAME = 'pathnex-web-app'
    }
    stages{
        stage{
            steps {
                script {
                    docker.build("${env.Image_Name})
                }
            }
        }
        stage('Psh Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image("{env.IMAGE_NAME}").push('latest')
                    }
                }
            }
        }
        stage('Deploy to Kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}
