CI/CD Using Jenkins:
pipeline {															(Declarative Pipeline)(Starts the complete pipeline defination)
	agent any														(Run on any available agent)
	stages {														(Contain All stages)
		stage('Build') {												(First Stage named Build)
			steps {													(List of actions to execute)
				echo 'Building projects'									(Print the text in console)
			}
		}
		stage('Test') {													(Second Stage named Test)
			steps {													(List of actions to execute)
				echo 'Running tests'										(Print the text in console)
			}
		}
		stage('Deploy') {												(Third Stage named Deploy)
			steps {													(List of actions to execute)
				echo 'Deploying application'									(Print the text in console)
			}
		}
	}
}


GitLab CI/CD - Basic Pipeline
stages:																(Keyword that list the phases of the pipeline)
  - build															(Phase 1)
  - test															(Phase 2)
  - deploy															(Phase 3)

build:																(Name of the Job)
  stage: build															(Stage assignment)
  script:															(Keyword)
    - echo "Building project"													(Script/Shell Commands)
	
test:																(Name of the Job)
  stage: test															(Stage assignment)
  script:															(Keyword)
    - echo "Running tests"													(Script/Shell Commands)
	
deploy:																(Name of the Job)
	stage: deploy														(Stage assignment)
	scipt:															(Keyword)
	  - echo "Deploying Application"											(Script/Shell Commands)

Ansible:															(Playbooks)
- name: Install & Confiqure Nginx												(Just a name to define what we are going to do)
  hosts: all															(Servers)
  become: yes															(Root Privelage)
  tasks:															(An Individual task a playbook is going to do there can be multiple tasks)
    - name: Install Nginx													(To know what we are going to do)
	  yum:															(Package Installer)
	    name: nginx														(Name of package we want)
		state: present													(It have to be installed)
	- name: Configure nginx													(To know what we are going to do)
	  template:														(Module that copy a file to the server)
	    src: nginx.conf.j2													(Source file)
		dest: /etc/nginx/nginx.conf											(Distination)
	-name: Start nginx service												(To know what we are going to do)
	 service:														(Control Service)
	   name: nginx														(Service Name)
	   state: started													(It will start after the command)
	   enabled: yes														(Automatically start whenever server boots)
