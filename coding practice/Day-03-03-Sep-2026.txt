Ansible:
- name: Install Apache & Start Service                  (Just a name to define what we are going to do)
  hosts: all											(Servers connected to main i.e. webservers, dataservers etc)
  become: yes											(Root user Prevelage)
  tasks:												(An Individual task a playbook is going to do there can be multiple tasks)
    - name: Install apache								(To know what we are going to do)
	  yum:												(Package Manager)
	    name: httpd										(Package name like httpd, nginex)
		state: present									(Have to be install its mandetory)
	
    - name: Start Apache								(To know what we are going to do)
	  service:												(Package Manager)
	    name: httpd										(Tell Server which service to control)
		state: started									(Start Apache Now)	
		enabled: yes									(It automatecally starts whenever server boots)

Terraform:

  region = "us-east-1"									(Region detail)
}
														(aws_instance is type of resource)

  }
}

Kubernetes:
apiversion: apps/v1                                     (K8s API version used for this resource)
kind: ReplicaSet										(Type of resource i want to create)
metadata:												(Contain info about resource)
  name: pathnex-replicaset								(Just a name which can be anyone)
spec: 													(Desired setup will be under spec)
  replicas: 3											(3 pods for the package we want)
  selector:												(It tells which pod to manage)
    matchLables:										
	app: pathnex-app									(Appis tag key & pathnex-app is valve)
  template:												(Blueprint use to create pod)
    metadata:											
	  lables:											
	    app: pathnex-app								(Each pod Lable)
	spec:
	  containers:										(List the containers inside each pod)
	    - name: app										(Name of containers)
		  image: nginx									(Container image available open source)
		  ports:										
		    - containerport: 80							(Port the image is going to use)
