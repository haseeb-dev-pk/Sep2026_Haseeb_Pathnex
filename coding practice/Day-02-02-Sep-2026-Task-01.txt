-name: Install & Enable Nginx   			(This line is just for the Name so any one can understand what we are doing)
  hosts: all						(This line is for which server you want to do the job)
  become: yes						(Use to allow Ansible the root permissions)
							(Blank as YAML file need)
  tasks:						(An Individual job that Ansible have to do)
    - name: Install nginx				(A task name so we know what we are doing)
      yum:						(An Ansible Module used to run Package)
		name: nginx				(This tells Ansible which package we want write now we want nginx)
		state: present				(This Means that Make sure Nginx is Installed)
							(Blank as YAML file need)
    - name: Enable nginx				(Again the name we can identify what we are doing in that)
	  services:					(It wants to manage Linux Services)
	  name: nginx					(Which package we want to manage)
	  state: started				(it Means to Start Nginx Right now)
	  enabled: yes					(This means to autometicaly Start Nginx on boot)
